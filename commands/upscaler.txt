const { downloadContentFromMessage } = require('@whiskeysockets/baileys');
const axios = require('axios');
const FormData = require('form-data');
const fs = require('fs');
const path = require('path');

/**
 * Fungsi untuk download media dari pesan WhatsApp
 */
async function downloadMedia(message) {
    try {
        const stream = await downloadContentFromMessage(message, 'image');
        let buffer = Buffer.from([]);
        
        for await (const chunk of stream) {
            buffer = Buffer.concat([buffer, chunk]);
        }
        
        return buffer;
    } catch (error) {
        console.error('[Upscaler] Error downloading media:', error);
        throw error;
    }
}

/**
 * Fungsi untuk upscale menggunakan API Waifu2x (Free)
 */
async function upscaleWithWaifu2x(inputBuffer) {
    try {
        console.log('[Upscaler] Mencoba Waifu2x API...');
        
        const formData = new FormData();
        formData.append('image', inputBuffer, {
            filename: 'image.jpg',
            contentType: 'image/jpeg'
        });
        
        // API Waifu2x gratis yang masih aktif
        const response = await axios.post('https://api.waifu2x.booru.pics/api/v1/waifu2x', formData, {
            headers: {
                ...formData.getHeaders()
            },
            timeout: 60000,
            responseType: 'arraybuffer'
        });
        
        if (response.status === 200) {
            console.log('[Upscaler] Waifu2x API berhasil');
            return Buffer.from(response.data);
        } else {
            throw new Error('API response tidak valid');
        }
    } catch (error) {
        console.error('[Upscaler] Error Waifu2x API:', error.message);
        throw error;
    }
}

/**
 * Fungsi untuk upscale menggunakan API Replicate (Free tier)
 */
async function upscaleWithReplicate(inputBuffer) {
    try {
        console.log('[Upscaler] Mencoba Replicate API...');
        
        // Convert buffer to base64
        const base64Image = `data:image/jpeg;base64,${inputBuffer.toString('base64')}`;
        
        const response = await axios.post('https://api.replicate.com/v1/predictions', {
            version: "30c1d0b916a6f8efce669b602ad59e38d6824363400c0ee3b93a54b373f78456",
            input: {
                image: base64Image,
                scale: 2
            }
        }, {
            headers: {
                'Authorization': 'Token YOUR_REPLICATE_API_TOKEN', // Ganti dengan token Anda
                'Content-Type': 'application/json'
            },
            timeout: 60000
        });
        
        if (response.data && response.data.urls && response.data.urls.get) {
            // Poll untuk hasil
            const resultUrl = response.data.urls.get;
            let result = null;
            
            for (let i = 0; i < 30; i++) { // Max 30 attempts (5 menit)
                await new Promise(resolve => setTimeout(resolve, 10000)); // Wait 10 seconds
                
                const resultResponse = await axios.get(resultUrl, {
                    headers: {
                        'Authorization': 'Token YOUR_REPLICATE_API_TOKEN'
                    }
                });
                
                if (resultResponse.data.status === 'succeeded') {
                    result = resultResponse.data.output;
                    break;
                } else if (resultResponse.data.status === 'failed') {
                    throw new Error('Replicate processing failed');
                }
            }
            
            if (result) {
                const imageResponse = await axios.get(result, {
                    responseType: 'arraybuffer'
                });
                
                console.log('[Upscaler] Replicate API berhasil');
                return Buffer.from(imageResponse.data);
            } else {
                throw new Error('Timeout waiting for result');
            }
        } else {
            throw new Error('Invalid response from Replicate API');
        }
    } catch (error) {
        console.error('[Upscaler] Error Replicate API:', error.message);
        throw error;
    }
}

/**
 * Fungsi untuk validasi gambar tanpa Sharp
 */
async function validateImage(buffer, maxSize = 10 * 1024 * 1024) {
    try {
        if (buffer.length > maxSize) {
            throw new Error(`Ukuran file terlalu besar. Maksimal ${Math.round(maxSize / 1024 / 1024)}MB`);
        }
        
        // Validasi format gambar dengan magic bytes
        const isJPEG = buffer[0] === 0xFF && buffer[1] === 0xD8;
        const isPNG = buffer[0] === 0x89 && buffer[1] === 0x50 && buffer[2] === 0x4E && buffer[3] === 0x47;
        const isWEBP = buffer[8] === 0x57 && buffer[9] === 0x45 && buffer[10] === 0x42 && buffer[11] === 0x50;
        const isGIF = buffer[0] === 0x47 && buffer[1] === 0x49 && buffer[2] === 0x46;
        
        if (!isJPEG && !isPNG && !isWEBP && !isGIF) {
            throw new Error('Format gambar tidak didukung. Gunakan JPG, PNG, WEBP, atau GIF');
        }
        
        // Estimasi ukuran berdasarkan file size
        let estimatedWidth = 1000;
        let estimatedHeight = 1000;
        
        if (buffer.length < 100 * 1024) { // < 100KB
            estimatedWidth = 500;
            estimatedHeight = 500;
        } else if (buffer.length > 1024 * 1024) { // > 1MB
            estimatedWidth = 1500;
            estimatedHeight = 1500;
        }
        
        return {
            width: estimatedWidth,
            height: estimatedHeight,
            format: isJPEG ? 'jpeg' : isPNG ? 'png' : isWEBP ? 'webp' : 'gif',
            size: buffer.length
        };
    } catch (error) {
        throw error;
    }
}

/**
 * Fungsi utama untuk processing upscale
 */
async function processUpscale(inputBuffer, scale = 2) {
    try {
        // Validasi gambar
        const metadata = await validateImage(inputBuffer);
        console.log(`[Upscaler] Gambar tervalidasi: ${metadata.width}x${metadata.height} (${metadata.format})`);
        
        let upscaledBuffer;
        let method = 'unknown';
        let lastError = null;
        
        // Coba berbagai API secara berurutan (yang gratis dan tidak perlu API key)
        const apis = [
            { name: 'Waifu2x', func: upscaleWithWaifu2x },
            // Uncomment yang di bawah setelah mendapatkan API key
            // { name: 'DeepAI', func: upscaleWithDeepAI },
            // { name: 'Clipdrop', func: upscaleWithClipdrop },
            // { name: 'Replicate', func: upscaleWithReplicate }
        ];
        
        for (const api of apis) {
            try {
                console.log(`[Upscaler] Mencoba ${api.name}...`);
                upscaledBuffer = await api.func(inputBuffer, scale);
                method = api.name.toLowerCase().replace(/[^a-z]/g, '');
                console.log(`[Upscaler] ${api.name} berhasil!`);
                break;
            } catch (error) {
                console.log(`[Upscaler] ${api.name} gagal: ${error.message}`);
                lastError = error;
                continue;
            }
        }
        
        // Jika semua API gagal, return gambar asli
        if (!upscaledBuffer) {
            console.log('[Upscaler] Semua API gagal, mengembalikan gambar asli...');
            upscaledBuffer = inputBuffer;
            method = 'original';
        }
        
        return {
            buffer: upscaledBuffer,
            originalSize: inputBuffer.length,
            resultSize: upscaledBuffer.length,
            originalRes: `${metadata.width}x${metadata.height}`,
            method: method,
            scale: upscaledBuffer === inputBuffer ? 1 : scale,
            success: upscaledBuffer !== inputBuffer
        };
    } catch (error) {
        console.error('[Upscaler] Error processing upscale:', error);
        throw error;
    }
}
